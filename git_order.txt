git常见命令

git init            初始化仓库
git clone <url>     克隆远程仓库
git status          查看仓库状态
git remove -v       查看远程仓库信息

git add <file>	        添加特定文件到暂存区
git add .	            添加所有修改到暂存区
git commit -m "消息"	提交更改
git commit --amend	    修改最新提交信息
git reset HEAD <file>	从暂存区移除文件
git checkout -- <file>	丢弃工作区的修改


git branch	                查看本地分支
git branch -a	            查看所有分支（含远程）
git branch <name>	        创建新分支
git checkout <branch>	    切换到分支
git checkout -b <branch>	创建并切换到新分支
git merge <branch>	        合并分支到当前分支
git branch -d <branch>	    删除已合并的分支
git branch -D <branch>	    强制删除分支

git fetch	                        下载远程更新但不合并
git pull	                        下载并合并远程更新
git push	                        推送本地提交到远程
git push -u origin <branch>	        推送并设置上游分支
git push origin --delete <branch>	删除远程分支
git remote show origin	            查看远程仓库详情
